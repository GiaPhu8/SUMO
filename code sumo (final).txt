/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{

  /* USER CODE BEGIN 1 */

  /* USER CODE END 1 */

  /* MCU Configuration--------------------------------------------------------*/

  /* Reset of all peripherals, Initializes the Flash interface and the Systick. */
  HAL_Init();

  /* USER CODE BEGIN Init */

  /* USER CODE END Init */

  /* Configure the system clock */
  SystemClock_Config();

  /* USER CODE BEGIN SysInit */

  /* USER CODE END SysInit */

  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_ADC1_Init();
  MX_TIM1_Init();
  MX_TIM2_Init();
  MX_I2C1_Init();
  MX_TIM3_Init();
  MX_TIM4_Init();
  /* USER CODE BEGIN 2 */

HAL_ADCEx_Calibration_Start(&hadc1);

HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_1);
HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_2);

HAL_TIM_Base_Start(&htim2);
// Khởi động MPU6050
MPU6050_Init();

HAL_Delay(200);

// Robot PHẢI đứng yên trong lúc calibration
MPU6050_CalibrateGyroZ();
// Mới cấp nguồn: LED tắt, robot đứng yên
HAL_GPIO_WritePin(GPIOB, GPIO_PIN_14, GPIO_PIN_RESET);
Motor_Stop();
rotateActive = 0;
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */
  // =====================================================
// 1. BUTTON ON / OFF
// PB15 INPUT_PULLUP
// Nhấn = LOW
// =====================================================
if (HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_15) == GPIO_PIN_RESET)
{
    HAL_Delay(50);

    if (HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_15) == GPIO_PIN_RESET)
    {
        // Đảo trạng thái ON/OFF
        systemActive = !systemActive;

        // ===============================
        // BẬT HỆ THỐNG
        // ===============================
        if (systemActive == 1)
        {
            // LED sáng
            HAL_GPIO_WritePin(
                GPIOB,
                GPIO_PIN_14,
                GPIO_PIN_SET
            );

            Motor_Stop();

            // Chưa quay
            rotateActive = 0;

            // Bắt đầu đếm 3 giây
            noEnemyStart = HAL_GetTick();
        }

        // ===============================
        // TẮT HỆ THỐNG
        // ===============================
        else
        {
            // LED tắt
            HAL_GPIO_WritePin(
                GPIOB,
                GPIO_PIN_14,
                GPIO_PIN_RESET
            );

            Motor_Stop();

            rotateActive = 0;
        }

        // Chờ nhả nút
        while (HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_15)
               == GPIO_PIN_RESET)
        {
        }

        HAL_Delay(50);
    }
}


// =====================================================
// 2. NẾU HỆ THỐNG ĐANG TẮT
// =====================================================
if (systemActive == 0)
{
    Motor_Stop();

    HAL_GPIO_WritePin(
        GPIOB,
        GPIO_PIN_14,
        GPIO_PIN_RESET
    );

    HAL_Delay(10);

    continue;
}


// =====================================================
// 3. HỆ THỐNG ON
// =====================================================
HAL_GPIO_WritePin(
    GPIOB,
    GPIO_PIN_14,
    GPIO_PIN_SET
);


// =====================================================
// 4. ĐỌC HC-SR04 PHÍA SAU
// PB10 = TRIG
// PB11 = ECHO
// =====================================================
distanceLeft = Read_HCSR04(
    GPIOB,
    GPIO_PIN_10,
    GPIOB,
    GPIO_PIN_11
);


// Tránh 2 HC-SR04 nhiễu nhau
HAL_Delay(30);


// =====================================================
// 5. ĐỌC HC-SR04 PHÍA TRƯỚC
// PB12 = TRIG
// PB13 = ECHO
// =====================================================
distanceRight = Read_HCSR04(
    GPIOB,
    GPIO_PIN_12,
    GPIOB,
    GPIO_PIN_13
);


// =====================================================
// 6. KIỂM TRA ĐỐI THỦ
// =====================================================
uint8_t enemyRear = 0;
uint8_t enemyFront = 0;


// HC phía sau thấy
if ((distanceLeft > 2) &&
    (distanceLeft <= OBSTACLE_DISTANCE))
{
    enemyRear = 1;
}


// HC phía trước thấy
if ((distanceRight > 2) &&
    (distanceRight <= OBSTACLE_DISTANCE))
{
    enemyFront = 1;
}


// =====================================================
// 7. CÓ ĐỐI THỦ
// -> HỦY QUAY VÀ LAO MAXIMUM
// =====================================================
if ((enemyFront == 1) ||
    (enemyRear == 1))
{
    // Nếu đang quay 1080° thì HỦY NGAY
    rotateActive = 0;


    // =============================================
    // HC TRƯỚC THẤY
    // -> TIẾN MAXIMUM
    // =============================================
    if ((enemyFront == 1) &&
        (enemyRear == 0))
    {
        Motor_Forward(
            SPEED_ATTACK,
            SPEED_ATTACK
        );
    }


    // =============================================
    // HC SAU THẤY
    // -> LÙI MAXIMUM
    // =============================================
    else if ((enemyRear == 1) &&
             (enemyFront == 0))
    {
        Motor_Backward(
            SPEED_ATTACK,
            SPEED_ATTACK
        );
    }


    // =============================================
    // CẢ HAI CÙNG THẤY
    // -> chọn đối thủ gần hơn
    // =============================================
    else
    {
        // Phía trước gần hơn
        if (distanceRight <= distanceLeft)
        {
            Motor_Forward(
                SPEED_ATTACK,
                SPEED_ATTACK
            );
        }

        // Phía sau gần hơn
        else
        {
            Motor_Backward(
                SPEED_ATTACK,
                SPEED_ATTACK
            );
        }
    }


    // Khi còn thấy đối thủ
    // liên tục reset thời gian 3 giây
    noEnemyStart = HAL_GetTick();
}


// =====================================================
// 8. CẢ HAI HC-SR04 KHÔNG THẤY ĐỐI THỦ
// =====================================================
else
{
    // =============================================
    // ĐANG QUAY 1080°
    // =============================================
    if (rotateActive == 1)
    {
        // Tiếp tục cập nhật góc bằng MPU6050
        if (Update_360_Rotation() == 1)
        {
            // Đã quay đủ 1080°
            Motor_Stop();

            rotateActive = 0;

            // Quay xong thì bắt đầu
            // đếm lại 3 giây
            noEnemyStart = HAL_GetTick();
        }
    }


    // =============================================
    // CHƯA QUAY
    // =============================================
    else
    {
        // Chưa đủ 3 giây
        if ((HAL_GetTick() - noEnemyStart)
            < NO_ENEMY_WAIT_TIME)
        {
            // Đứng yên chờ
            Motor_Stop();
        }

        // Đủ 3 giây vẫn không thấy đối thủ
        else
        {
            // Bắt đầu quay 1080°
            Start_360_Rotation();

            // Chạy lần cập nhật đầu tiên
            Update_360_Rotation();
        }
    }
}
  /* USER CODE END 3 */
}
}

// MPU6050 INIT

void MPU6050_Init(void)
{
    uint8_t data;

    // Wake up MPU6050
    data = 0x00;

    HAL_I2C_Mem_Write(
        &hi2c1,
        MPU6050_ADDR,
        MPU6050_PWR_MGMT_1,
        I2C_MEMADD_SIZE_8BIT,
        &data,
        1,
        100
    );

    HAL_Delay(100);


    // Gyroscope ±500 degree/s
    // sensitivity = 65.5 LSB / degree/s
    data = 0x08;

    HAL_I2C_Mem_Write(
        &hi2c1,
        MPU6050_ADDR,
        MPU6050_GYRO_CONFIG,
        I2C_MEMADD_SIZE_8BIT,
        &data,
        1,
        100
    );
}


// =====================================================
// ĐỌC GYRO TRỤC Z
// =====================================================
float MPU6050_ReadGyroZ(void)
{
    uint8_t data[2];

    int16_t gyroRaw;

    HAL_I2C_Mem_Read(
        &hi2c1,
        MPU6050_ADDR,
        MPU6050_GYRO_ZOUT_H,
        I2C_MEMADD_SIZE_8BIT,
        data,
        2,
        100
    );


    gyroRaw =
        (int16_t)((data[0] << 8) | data[1]);


    // ±500 degree/s
    // sensitivity = 65.5
    return ((float)gyroRaw / 65.5f) - gyroZOffset;
}


// =====================================================
// CALIBRATE GYRO Z
// ROBOT PHẢI ĐỨNG YÊN
// =====================================================
void MPU6050_CalibrateGyroZ(void)
{
    uint16_t i;

    float sum = 0.0f;

    uint8_t data[2];

    int16_t gyroRaw;


    for (i = 0; i < 500; i++)
    {
        HAL_I2C_Mem_Read(
            &hi2c1,
            MPU6050_ADDR,
            MPU6050_GYRO_ZOUT_H,
            I2C_MEMADD_SIZE_8BIT,
            data,
            2,
            100
        );


        gyroRaw =
            (int16_t)((data[0] << 8) | data[1]);


        sum += (float)gyroRaw / 65.5f;


        HAL_Delay(2);
    }


    gyroZOffset = sum / 500.0f;
}


// =====================================================
// BẮT ĐẦU QUAY 360 ĐỘ
// =====================================================
void Start_360_Rotation(void)
{
    rotateAngle = 0.0f;

    rotateStartTime = HAL_GetTick();
    gyroLastTime = HAL_GetTick();

    rotateActive = 1;
}


// =====================================================
// CẬP NHẬT QUÁ TRÌNH QUAY 360 ĐỘ
//
// return:
// 0 = chưa đủ 360 độ
// 1 = đã đủ 360 độ
// =====================================================
uint8_t Update_360_Rotation(void)
{
    uint32_t now;

    uint32_t elapsed;

    float dt;

    float gyroZ;

    float desiredAngle;

    float error;

    int32_t rotateSpeed;


    now = HAL_GetTick();


    // ---------------------------------------------
    // Tính thời gian giữa 2 lần đọc gyro
    // ---------------------------------------------
    dt =
        (float)(now - gyroLastTime)
        / 1000.0f;

    gyroLastTime = now;


    // ---------------------------------------------
    // Đọc vận tốc góc
    // ---------------------------------------------
    gyroZ = MPU6050_ReadGyroZ();


    // Chỉ cần độ lớn
    if (gyroZ < 0.0f)
    {
        gyroZ = -gyroZ;
    }


    // ---------------------------------------------
    // Tính góc đã quay
    //
    // góc = vận tốc góc × thời gian
    // ---------------------------------------------
    rotateAngle += gyroZ * dt;


    // ---------------------------------------------
    // Đã đủ 360 độ
    // ---------------------------------------------
    if (rotateAngle >= ROTATE_TARGET_ANGLE)
    {
        Motor_Stop();

        rotateActive = 0;

        return 1;
    }


    // ---------------------------------------------
    // Điều khiển để cố gắng hoàn thành trong 3s
    // ---------------------------------------------
    elapsed =
        now - rotateStartTime;
// =============================================
// AN TOÀN:
// nếu quá 5 giây vẫn chưa đủ 360 độ
// -> dừng motor để tránh quay vô hạn
// =============================================
if (elapsed >= ROTATE_MAX_TIME)
{
    Motor_Stop();
    rotateActive = 0;

    return 1;
}

    desiredAngle =
        ROTATE_TARGET_ANGLE *
        ((float)elapsed /
        (float)ROTATE_TARGET_TIME);


    if (desiredAngle > ROTATE_TARGET_ANGLE)
    {
        desiredAngle = ROTATE_TARGET_ANGLE;
    }


    // Sai số góc
    error =
        desiredAngle - rotateAngle;


    // PWM cơ bản = 500
    // cộng hiệu chỉnh dựa trên sai số
    rotateSpeed =
        SPEED_SEARCH +
        (int32_t)(error * 3.0f);


    // Không cho quá chậm
    if (rotateSpeed < 250)
    {
        rotateSpeed = 250;
    }


    // Không vượt PWM max
    if (rotateSpeed > 999)
    {
        rotateSpeed = 999;
    }


    Motor_Rotate_Left(
        (uint16_t)rotateSpeed
    );


    return 0;
}
